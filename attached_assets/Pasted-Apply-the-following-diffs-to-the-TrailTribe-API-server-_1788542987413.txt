+// Global error handler — registered last, after all routes. Express falls
+// back to its own default error handler for anything unhandled below this
+// point, and that default's behavior (whether it includes a stack trace)
+// depends on NODE_ENV. This repo doesn't set NODE_ENV=production anywhere
+// in its own scripts, so that default is an implicit dependency on the
+// deploy platform rather than a guarantee. This handler removes that
+// dependency: it never returns internal details to the client regardless
+// of environment, while still logging the full error server-side.
+// eslint-disable-next-line @typescript-eslint/no-unused-vars
+app.use((err: any, req: express.Request, res: express.Response, _next: express.NextFunction) => {
+  req.log?.error({ err }, "Unhandled error") ?? logger.error({ err }, "Unhandled error");
+
+  if (res.headersSent) {
+    return;
+  }
+
+  const status = typeof err?.status === "number" ? err.status
+    : typeof err?.statusCode === "number" ? err.statusCode
+    : 500;
+
+  res.status(status).json({ error: status >= 500 ? "Internal server error" : (err?.message || "Request failed") });
+});
+
 export default app;
